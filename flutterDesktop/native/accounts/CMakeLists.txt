cmake_minimum_required(VERSION 3.16)
project(bitchord_accounts LANGUAGES CXX)
add_executable(bitchord_secrets secret_store.cpp)
target_compile_features(bitchord_secrets PRIVATE cxx_std_17)
if(WIN32)
  target_link_libraries(bitchord_secrets PRIVATE crypt32)
  target_compile_definitions(bitchord_secrets PRIVATE UNICODE _UNICODE)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(BC_SECRET REQUIRED IMPORTED_TARGET libsecret-1)
  target_link_libraries(bitchord_secrets PRIVATE PkgConfig::BC_SECRET)
endif()
