cmake_minimum_required(VERSION 3.16)
project(bitchord_audio LANGUAGES CXX)
option(BC_TESTS "Build native contract tests" ON)
option(BC_SANITIZE "Address/undefined behavior instrumentation" OFF)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../accounts" "${CMAKE_CURRENT_BINARY_DIR}/accounts")
add_library(bitchord_audio SHARED src/engine.cpp src/decoder.cpp src/store.cpp src/provider_crypto.cpp)
target_compile_features(bitchord_audio PRIVATE cxx_std_17)
target_compile_definitions(bitchord_audio PRIVATE BC_BUILD)
target_include_directories(bitchord_audio PUBLIC include PRIVATE src)
set_target_properties(bitchord_audio PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(WIN32)
  target_sources(bitchord_audio PRIVATE src/sink_windows.cpp)
  find_path(BC_FFMPEG_INCLUDE libavformat/avformat.h REQUIRED)
  target_include_directories(bitchord_audio PRIVATE ${BC_FFMPEG_INCLUDE})
  foreach(component avformat avcodec avutil swresample)
    find_library(BC_${component} NAMES ${component} REQUIRED)
    target_link_libraries(bitchord_audio PRIVATE ${BC_${component}})
  endforeach()
  find_package(unofficial-sqlite3 CONFIG REQUIRED)
  target_link_libraries(bitchord_audio PRIVATE unofficial::sqlite3::sqlite3 ole32 uuid avrt)
  target_compile_definitions(bitchord_audio PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(BC_DEPS REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswresample libpulse alsa sqlite3)
  target_sources(bitchord_audio PRIVATE src/sink_linux.cpp)
  target_link_libraries(bitchord_audio PRIVATE PkgConfig::BC_DEPS pthread)
endif()
if(BC_SANITIZE AND NOT MSVC)
  target_compile_options(bitchord_audio PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bitchord_audio PRIVATE -fsanitize=address,undefined)
endif()
add_executable(bitchord_audio_probe test/probe.cpp)
target_compile_features(bitchord_audio_probe PRIVATE cxx_std_17)
target_link_libraries(bitchord_audio_probe PRIVATE bitchord_audio)
set_target_properties(bitchord_audio_probe PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
if(BC_TESTS)
  enable_testing()
  add_executable(bitchord_audio_test test/native_test.cpp)
  target_compile_features(bitchord_audio_test PRIVATE cxx_std_17)
  target_include_directories(bitchord_audio_test PRIVATE src)
  target_link_libraries(bitchord_audio_test PRIVATE bitchord_audio)
  add_executable(bitchord_decoder_test test/decoder_test.cpp src/decoder.cpp)
  target_compile_features(bitchord_decoder_test PRIVATE cxx_std_17)
  target_include_directories(bitchord_decoder_test PRIVATE src include)
  add_executable(bitchord_stream_test test/stream_test.cpp src/decoder.cpp)
  target_compile_features(bitchord_stream_test PRIVATE cxx_std_17)
  target_include_directories(bitchord_stream_test PRIVATE src include)
  if(WIN32)
    target_include_directories(bitchord_stream_test PRIVATE ${BC_FFMPEG_INCLUDE})
    target_link_libraries(bitchord_stream_test PRIVATE ${BC_avformat} ${BC_avcodec} ${BC_avutil} ${BC_swresample})
    target_include_directories(bitchord_decoder_test PRIVATE ${BC_FFMPEG_INCLUDE})
    target_link_libraries(bitchord_decoder_test PRIVATE ${BC_avformat} ${BC_avcodec} ${BC_avutil} ${BC_swresample})
  else()
    target_link_libraries(bitchord_stream_test PRIVATE PkgConfig::BC_DEPS pthread)
    target_link_libraries(bitchord_decoder_test PRIVATE PkgConfig::BC_DEPS pthread)
  endif()
  if(BC_SANITIZE AND NOT MSVC)
    foreach(t bitchord_audio_test bitchord_audio_probe bitchord_decoder_test bitchord_stream_test)
      target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
  add_test(NAME native_decoder COMMAND bitchord_decoder_test "${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures")
  add_test(NAME native_contract COMMAND bitchord_audio_test)
endif()
